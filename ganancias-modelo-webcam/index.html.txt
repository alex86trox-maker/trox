function loadCardToForm(i){
  const c = edCards[i];
  document.getElementById('ep-c-tag').value    = c.tag;
  document.getElementById('ep-c-title').value  = c.title;
  document.getElementById('ep-c-time').value   = c.time;
  document.getElementById('ep-c-income').value = c.income;
  document.getElementById('ep-c-period').value = c.period;
  document.getElementById('ep-c-hl').checked   = c.highlight;
  renderListEditor(i);
  // live bind
  ['tag','title','time','income','period'].forEach(f => {
    const el = document.getElementById('ep-c-'+f);
    el.oninput = () => { edCards[activeCard][f] = el.value; };
  });
  document.getElementById('ep-c-hl').onchange = e => { edCards[activeCard].highlight = e.target.checked; };
}
function renderListEditor(i){
  const c = edCards[i];
  const cont = document.getElementById('ep-list');
  cont.innerHTML = '';
  c.items.forEach((item, idx) => {
    const row = document.createElement('div');
    row.className = 'list-row';
    const inp = document.createElement('input');
    inp.type = 'text'; inp.className = 'ep-input'; inp.value = item;
    inp.oninput = () => { c.items[idx] = inp.value; };
    const rm = document.createElement('button');
    rm.className = 'btn-rm'; rm.innerHTML = '×';
    rm.onclick = () => { c.items.splice(idx,1); renderListEditor(activeCard); };
    row.appendChild(inp); row.appendChild(rm);
    cont.appendChild(row);
  });
  const add = document.createElement('button');
  add.className = 'btn-add'; add.innerHTML = '+ Agregar punto';
  add.onclick = () => { c.items.push('Nuevo punto'); renderListEditor(activeCard); };
  cont.appendChild(add);
}

function applyAll(){
  // read inputs first
  edSection.label   = document.getElementById('ep-label').value;
  edSection.title   = document.getElementById('ep-title').value;
  edSection.titleEm = document.getElementById('ep-title-em').value;
  edSection.desc    = document.getElementById('ep-desc').value;
  edCards[activeCard].tag     = document.getElementById('ep-c-tag').value;
  edCards[activeCard].title   = document.getElementById('ep-c-title').value;
  edCards[activeCard].time    = document.getElementById('ep-c-time').value;
  edCards[activeCard].income  = document.getElementById('ep-c-income').value;
  edCards[activeCard].period  = document.getElementById('ep-c-period').value;
  edCards[activeCard].highlight = document.getElementById('ep-c-hl').checked;

  // update section intro
  document.getElementById('rng-label').textContent = edSection.label;
  const titleEl = document.getElementById('rng-title-el');
  titleEl.childNodes[0].textContent = edSection.title + '\n';
  document.getElementById('rng-title-em').textContent = edSection.titleEm;
  document.getElementById('rng-desc').textContent = edSection.desc;

  // update cards
  edCards.forEach((c, i) => {
    const card = document.getElementById('ncard-'+i);
    if(!card) return;
    card.className = 'nivel-card reveal on' + (c.highlight ? ' highlight' : '');
    if(i===1) card.style.transitionDelay = '.1s';
    if(i===2) card.style.transitionDelay = '.2s';
    document.getElementById('nc'+i+'-tag').textContent    = c.tag;
    document.getElementById('nc'+i+'-title').textContent  = c.title;
    document.getElementById('nc'+i+'-time').textContent   = c.time;
    document.getElementById('nc'+i+'-income').textContent = c.income;
    document.getElementById('nc'+i+'-period').textContent = c.period;
    const list = document.getElementById('nc'+i+'-list');
    list.innerHTML = c.items.map(it => '<li>'+escH(it)+'</li>').join('');
  });
  showToast('Cambios aplicados');
}

function escH(s){ return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }

/* =============================================
   GENERATE CODE
============================================= */
function generateCode(){
  const s = edSection;
  const cards = edCards.map(c => {
    const items = c.items.map(it => '          <li>'+escH(it)+'</li>').join('\n');
    return `
      <div class="nivel-card${c.highlight?' highlight':''}">
        <div class="nivel-tag">${escH(c.tag)}</div>
        <div class="nivel-title">${escH(c.title)}</div>
        <div class="nivel-time">${escH(c.time)}</div>
        <div class="nivel-income">${escH(c.income)}</div>
        <div class="nivel-period">${escH(c.period)}</div>
        <div class="nivel-sep"></div>
        <ul class="nivel-list">
${items}
        </ul>
      </div>`;
  }).join('\n');
  return `<section class="rangos-bg section">
  <div class="container">
    <div class="rangos-intro">
      <span class="s-label reveal">${escH(s.label)}</span>
      <h2 class="s-title reveal" style="transition-delay:.1s">${escH(s.title)}<br><em>${escH(s.titleEm)}</em></h2>
      <p class="reveal" style="transition-delay:.2s">${escH(s.desc)}</p>
    </div>
    <div class="nivel-grid">
${cards}
    </div>
  </div>
</section>`;
}
function renderCode(){ document.getElementById('ep-code').textContent = generateCode(); }
function copyCode(){
  navigator.clipboard.writeText(generateCode())
    .then(() => showToast('Código copiado'))
    .catch(() => showToast('No se pudo copiar'));
}
function downloadCode(){
  const blob = new Blob([generateCode()], {type:'text/html'});
  const a = document.createElement('a'); a.href = URL.createObjectURL(blob);
  a.download = 'rangos-ganancias.html'; a.click(); URL.revokeObjectURL(a.href);
  showToast('Archivo descargado');
}
