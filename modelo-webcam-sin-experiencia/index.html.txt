function parallaxEl(el, section, speed){
  if (!el || !section) return;
  var r = section.getBoundingClientRect(), wh = window.innerHeight;
  if (r.bottom < 0 || r.top > wh) return;
  var v = (((wh - r.top) / (wh + r.height)) - 0.5) * speed;
  el.style.transform = 'translate3d(0,' + v + 'px,0)';
}
